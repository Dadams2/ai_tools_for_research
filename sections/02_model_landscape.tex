% ---- Capability tiers --------------------------------------
\begin{frame}{Capability Tiers}
  \begin{columns}[T, onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Chat / Instruction Models}
        \begin{itemize}
          \item Fast, conversational, general-purpose
          \item Strong instruction following
          \item \textbf{Best for:} writing, summarization,\newline Q\&A, coding assistance
          \item \textit{Examples:} \model{claude-sonnet-4-6}, \model{claude-haiku-4-5},\newline \model{gpt-5.5}, \model{gemini-3-flash}
        \end{itemize}
      \end{block}
    \end{column}
    \hfill
    \begin{column}{0.48\textwidth}
      \begin{block}{Reasoning / Thinking Models}
        \begin{itemize}
          \item Slower; internally ``thinks'' before responding
          \item Better at multi-step logical problems
          \item \textbf{Best for:} math proofs, algorithm design,\newline complex debugging, formal reasoning
          \item \textit{Examples:} \model{gpt-5.5} (reasoning config),\newline \model{claude-opus-4-8} (extended thinking),\newline \model{kimi-k2-thinking}, \model{deepseek-r1}
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \alert{Rule of thumb:} Start with a fast model; escalate to a reasoning model when the problem requires structured multi-step inference.
\end{frame}

      \textbf{Key benchmarks}\\[0.3em]
      \begin{itemize}
        \item \textbf{MMLU}~\parencite{hendrycks2021mmlu}: broad knowledge (57 subjects)
        \item \textbf{HumanEval}~\parencite{chen2021codex}: Python function synthesis
        \item \textbf{SWE-Bench}~\parencite{jimenez2024swebench}: real GitHub issue resolution
      \end{itemize}
      \vspace{0.6em}
      \textbf{Caveats}\\[0.3em]
      \begin{itemize}
        \item \textbf{Harness-dependence}: Agent benchmark scores (SWE-bench, Terminal-Bench) measure the \textbf{model + harness + tools + budget}, not model alone. Scaffolding dominates results.
        \item Contamination risk — benchmarks leak into training data~\parencite{srivastava2022bigbench}
        \item Metric design can favor verbose or multi-rollout structures
      \end{itemize}
    \end{column}

% ---- Local Agentic Workflows -------------------------------
\begin{frame}{Local Models: From Chat to Agentic Workflows}
  \begin{columns}[T, onlytextwidth]
    \begin{column}{0.48\textwidth}
      \textbf{The local paradigm shift}
      \begin{itemize}
        \item Moved beyond simple ``cheap chat models on Ollama''
        \item Edge intelligence enables private, continuous run loops
      \end{itemize}
      \vspace{0.4em}
      \textbf{Local models are now useful for:}
      \begin{itemize}
        \item Private codebase Q\&A and simple refactors
        \item Offline/air-gapped drafting and summarization
        \item Edge multimodal agents (laptops/devices)
        \item Low-cost batch extraction operations
        \item Custom fine-tuning and distillation pipelines
      \end{itemize}
    \end{column}
    \hfill
    \begin{column}{0.48\textwidth}
      \textbf{Deployment and Execution}
      \begin{itemize}
        \item Run fully local with \tool{Ollama}, \tool{LM Studio}, or \tool{vLLM}
        \item Decoupled private APIs for academic environments
      \end{itemize}
      \vspace{0.4em}
      \begin{block}{Security Advantage}
        Zero data leakage to external commercial servers. Ensures complete IP custody of unpublished research, drafts, and experimental results.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.4em}
  \centering\tiny\textit{All models updated as of June 2026.}
\end{frame}

% ---- Open-weight models ------------------------------------
\begin{frame}[fragile]{Leading Open-Weight Families}
  \begin{columns}[T, onlytextwidth]
    \begin{column}{0.55\textwidth}
      \textbf{Model Families \& Capabilities}\\[0.3em]
      \begin{itemize}\setlength{\itemsep}{1pt}
        \item \tool{Gemma 4 12B}: Mid-sized, native multimodal audio/vision, designed for agentic tasks on laptops.
        \item \tool{Gemma 3 / 3n / FunctionGemma}: Vision-language support, 128K context window, structured output/function calling.
        \item \tool{Kimi K2 / K2 Thinking}: 1T-total / 32B-active MoE; reasoning-capable open agentic workflows.
        \item \tool{Qwen3.5}: Qwen3.5-Plus supports 1M context with built-in tools.
        \item \tool{Mistral Large 3}~\parencite{jiang2023mistral}: Multimodal, general-purpose MoE.
        \item \tool{Llama 4 Scout/Maverick}: Multimodal; Scout marketed around extremely long context (caveat benchmark comparison).
      \end{itemize}
    \end{column}
    \hfill
    \begin{column}{0.42\textwidth}
      \textbf{Local deployment}\\[0.3em]
      Run open-weight models with \tool{Ollama} / \tool{vLLM}:
      \begin{lstlisting}[language=bash, numbers=none, basicstyle=\ttfamily\tiny]
# Install Ollama and pull Gemma
ollama pull gemma4:12b

# Run local inference
ollama run gemma4:12b
      \end{lstlisting}
      \vspace{0.4em}
      \begin{itemize}
        \item Excellent for confidential draft synthesis and coding pipelines.
        \item Support via Aider/OpenCode with local endpoints.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.4em}
  \centering\tiny\textit{Model details as of June 2026. Verify benchmark comparability carefully.}
\end{frame}
